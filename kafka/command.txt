# Kafka — single-node setup with Zookeeper

# 1. Pull both Confluent images
docker pull confluentinc/cp-zookeeper
docker pull confluentinc/cp-kafka

# 2. Create a private bridge network so the two containers can talk
docker network create kafka

# 3. Start Zookeeper
docker run -d --network=kafka --name=zookeeper \
    -e ZOOKEEPER_CLIENT_PORT=2181 \
    -e ZOOKEEPER_TICK_TIME=2000 \
    -p 2181:2181 \
    confluentinc/cp-zookeeper

# 4. Start Kafka, pointed at the Zookeeper container
docker run -d --network=kafka --name=kafka \
    -e KAFKA_ZOOKEEPER_CONNECT=zookeeper:2181 \
    -e KAFKA_ADVERTISED_LISTENERS=PLAINTEXT://localhost:9092 \
    -p 9092:9092 \
    confluentinc/cp-kafka
