# Moonlight — Windows firewall rules

# Allows the Moonlight game-streaming protocol through the Windows firewall.
# Run from an elevated `cmd.exe` (Run as Administrator).

netsh advfirewall firewall add rule ^
    name="GameStream UDP" dir=in protocol=udp ^
    localport=5353,47998-48010 action=allow

netsh advfirewall firewall add rule ^
    name="GameStream TCP" dir=in protocol=tcp ^
    localport=47984,47989,48010 action=allow
