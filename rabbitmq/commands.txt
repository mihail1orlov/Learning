# RabbitMQ — quick start

## One-shot ephemeral container

# Spins up RabbitMQ with the management UI on port 15672. Disappears
# when the container stops (`--rm`).
docker run -it --rm --name rabbitmq \
    -p 5672:5672 \
    -p 15672:15672 \
    rabbitmq:3-management

# Then open http://localhost:15672 (default login: guest / guest).

## docker-compose for persistent storage

# Reads RABBITMQ_DEFAULT_USER and RABBITMQ_DEFAULT_PASS from a `.env` file.
version: '2'

services:
  rabbitmq:
    image: rabbitmq:3-management-alpine
    container_name: moon_rabbitmq
    volumes:
      - rabbitmq_etc:/etc/rabbitmq/
      - rabbitmq_data:/var/lib/rabbitmq/
      - rabbitmq_logs:/var/log/rabbitmq/
    environment:
      - RABBITMQ_DEFAULT_USER=${RABBITMQ_DEFAULT_USER}
      - RABBITMQ_DEFAULT_PASS=${RABBITMQ_DEFAULT_PASS}
    ports:
      - 5672:5672      # AMQP
      - 15672:15672    # Management UI
