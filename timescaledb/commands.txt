## Drop into the running container and use psql

docker exec -it moon_timescaledb bash

# Inside the container:
psql --help
psql -U postgres

## psql session — basics

\du                   # list roles
create database test;
\l                    # list databases
\c test               # connect to `test`

CREATE TABLE students (id uuid, name varchar(80), sname varchar(80), age int);

insert into students (id, name, sname, age)
values (uuid_generate_v1(), 'John', 'Conor', 21);

select * from students;

\dt                   # list tables
\d students           # describe a table

create extension if not exists "uuid-ossp";
select uuid_generate_v1();

\q                    # quit

## Connect from a Linux client

sudo apt update
sudo apt install postgresql postgresql-contrib

psql -h localhost -p 5432 -U postgres

# `postgresql-client-common` may also be required on minimal images,
# though I'm not 100% sure when it's actually needed.
sudo apt install postgresql-client-common

## Run TimescaleDB in Docker

docker run -d --name timescaledb \
    -p 5432:5432 \
    -e POSTGRES_PASSWORD=postgres \
    timescale/timescaledb:latest-pg12

## Run Portainer alongside it

docker run -d -p 9000:9000 \
    --name=portainer --restart always \
    -v /var/run/docker.sock:/var/run/docker.sock \
    -v portainer_data:/data \
    portainer/portainer

## docker-compose with pgAdmin

# Reads POSTGRES_PASSWORD, PGADMIN_DEFAULT_EMAIL, PGADMIN_DEFAULT_PASSWORD
# from a `.env` file.
version: '2'
services:
    timescaledb:
        image: timescale/timescaledb:latest-pg12
        container_name: moon_timescaledb
        environment:
            - POSTGRES_PASSWORD=${POSTGRES_PASSWORD}
        volumes:
            - db-data:/var/lib/postgresql/data
        ports:
            - "5432:5432"

    pgadmin:
        image: dpage/pgadmin4
        container_name: moon_pgadmin
        environment:
            - PGADMIN_DEFAULT_EMAIL=${PGADMIN_DEFAULT_EMAIL}
            - PGADMIN_DEFAULT_PASSWORD=${PGADMIN_DEFAULT_PASSWORD}
        ports:
            - "5050:80"
        restart: always
        volumes:
            - pgadmin:/root/.pgadmin

volumes:
    db-data:
    pgadmin:
