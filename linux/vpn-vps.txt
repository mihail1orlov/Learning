## OpenVPN server — VPS hardening checklist

# 1. Create a non-root user
useradd -m -s /bin/bash myName
id myName

# 2. Set passwords
passwd               # change root password
passwd myUser        # change a specific user's password

# 3. Lock down sshd_config
#    PermitRootLogin no
#    Port 222            (move off 22 to silence most bots)

# 4. Throttle brute-force attempts
sudo apt install fail2ban

# 5. Set a sensible hostname
sudo hostnamectl set-hostname myHost
sudo vim /etc/hosts

## OpenVPN server — install in one command

# Generates /root/<user>.ovpn for the first client.
wget https://git.io/vpn -O openvpn-install.sh && bash openvpn-install.sh
