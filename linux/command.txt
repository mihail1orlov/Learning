# Linux — everyday commands cheat sheet

## Resource monitoring

free          # memory in KB
free -m       # memory in MB
free -g       # memory in GB
# Sample output of `free -m`:
# total        used        free      shared  buff/cache   available
# Mem:          32064        1808         273           3       29982       29809
# Swap:          8191           2        8189

uptime
# Sample output:
# 10:56:53 up 11:00,  1 user,  load average: 0.08, 0.10, 0.06

ps aux                       # all processes
ps aux | grep bluetooth      # filter
top                          # interactive
sudo apt install htop        # nicer top

# macOS doesn't have `free`; use:
top -l 1 -s 0 | grep PhysMem

## History

# Make every entry timestamped:
export HISTTIMEFORMAT='%F %T'

# Last 10 entries:
history | tail -n 10

# Re-run by number:
!1305

## Power

systemctl poweroff
systemctl list-units --type=service | grep 'net'

## Filesystem inspection

ls
ls -l
cd ~
cd ..

ls /dev/sd*       # raw block devices
lsblk             # tree view of disks
df -h             # mounted filesystems with sizes

# Read-only logs / configs
cat ~/.bashrc
less ~/.bashrc

## Permissions and ownership

# `chown` changes file/dir/link ownership.
sudo chown $(whoami) <file or directory>

# Octal digit reference (read=4, write=2, execute=1):
# 0  ---  no permissions
# 1  --x  execute
# 2  -w-  write
# 3  -wx  write + execute
# 4  r--  read
# 5  r-x  read + execute
# 6  rw-  read + write
# 7  rwx  read + write + execute

## Networking

# Get the public IP of this host:
curl https://api.ipify.org/

# SCP a file to a remote host (full guide:
#   https://linuxize.com/post/how-to-use-scp-command-to-securely-transfer-files/):
scp file.txt remote_username@10.10.0.2:/remote/directory

## Distribution info

lsb_release -a    # release codename, ID, version

## Useful tools

locate <name>     # find files / dirs
sudo updatedb     # refresh the locate index
mc                # Midnight Commander (TUI file manager)
jq                # JSON processor: `cat file.json | jq`
wget              # download files (or recursively mirror sites)
