## x11vnc as a systemd service

apt install x11vnc

# Create a password file at ~/.vnc/passwd
x11vnc -storepasswd

# Service unit
sudo vim /lib/systemd/system/x11vnc.service

# --- contents ---
# [Unit]
# Description=Start x11vnc server.
# After=multi-user.target
#
# [Service]
# Type=simple
# ExecStart=/usr/bin/x11vnc -auth guess -forever -loop -noxdamage -repeat \
#     -rfbauth /home/moon/.vnc/passwd -rfbport 5900 -shared
# Restart=on-failure
# RestartSec=3
#
# [Install]
# WantedBy=multi-user.target
# --- end ---

sudo systemctl daemon-reload
sudo systemctl enable x11vnc.service
