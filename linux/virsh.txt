## Pre-flight checks

# CPU virtualisation flags
lscpu | grep Virtualization
grep -E --color=auto 'vmx|svm|0xc0f' /proc/cpuinfo

# Kernel modules
lsmod | grep kvm

# libvirtd service
systemctl is-active libvirtd
systemctl status -l libvirt

## Networking

brctl show                       # Linux bridge state
# bridge name     bridge id              STP enabled     interfaces
# virbr0          8000.525400ad834a      yes             virbr0-nic / enp0s25

virsh net-dhcp-leases default    # leases handed out by libvirt's NAT
virsh iface-list --all
virsh iface-edit eth0
sudo ip l set eth0 down          # take an interface down
sudo ifconfig eth0 down          # legacy equivalent
ip l sh eth0                     # inspect

## Inventory

virsh list --all                 # all defined VMs
virsh pool-list --all            # storage pools
virsh net-list --all             # virtual networks

# Connect to a remote libvirt over SSH
virsh -c qemu+ssh://user@192.168.x.x/system

## Day-to-day VM control (inside `virsh --connect qemu:///system`)

virsh # start    test            # power on
virsh # suspend  test            # freeze
virsh # resume   test            # unfreeze
virsh # shutdown test            # graceful shutdown (equivalent of `shutdown -h now`)
virsh # destroy  test            # forced power off (does NOT delete the disk)
virsh # undefine test            # remove the VM definition (config file goes away)
virsh # list                     # active VMs
virsh # list --all               # everything

## Helpers

# Available osinfo IDs (handy for `--os-variant` in virt-install)
sudo apt install libosinfo-bin
osinfo-query os | grep -i ubuntu | cut -d ' ' -f -2

# All attached block devices (excluding loop = 7)
sudo lsblk -e7
