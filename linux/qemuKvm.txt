# QEMU / KVM — disk images and `virt-install` recipes

## Install

sudo apt install qemu-kvm libvirt-daemon-system libvirt-clients -y
sudo apt install virt-manager   # optional GUI

## Disk image manipulation with qemu-img

# Create an empty 3 GiB qcow2 image with metadata pre-allocated
qemu-img create -f qcow2 -o preallocation=metadata myDiskName.qcow2 3G

# Inspect / resize / drop a snapshot
sudo qemu-img info debian.qcow2
sudo qemu-img resize debian.qcow2 +4G
sudo qemu-img snapshot -d snapshot1 debian.qcow2

## Create a Windows 10 VM with virt-install

# `bus=virtio` (disks) and `model=virtio` (NIC) are the magic that makes
# I/O fast — both are required.
virt-install -n Windows10-6Gb-4cpu-testing -r 6048 --vcpus=4 \
    --arch=x86_64 \
    --disk /var/lib/libvirt/images/Wind10.qcow2,format=qcow2,bus=virtio \
    --disk /root/iso_disks/ru_windows_10.iso,device=cdrom \
    --disk /root/iso_disks/virtio-win-0.1.185.iso,device=cdrom \
    --network bridge:virbr0,model=virtio \
    --os-type=windows --os-variant=win10 \
    --vnc --vncport=5905 \
    --boot cdrom,hd,menu=on

## Create a small Ubuntu Server VM with virt-install

virt-install --name ubuntutest --ram 2048 \
    --disk path=/var/lib/libvirt/images/disk0.qcow2,size=10 \
    --vcpus 2 \
    --os-type linux \
    --os-variant generic \
    --console pty,target_type=serial \
    --graphics vnc \
    --cdrom /var/lib/libvirt/images/isos/ubuntu-18.04.6-live-server-amd64.iso

## Day-to-day virsh

# Image storage
ls /var/lib/libvirt/images

# Service health
systemctl status libvirtd

# Inventory
virsh list --all
virsh net-list
virsh net-edit default     # edit the default virtual network

# Per-VM control
virsh start    <vm-name>
virsh reboot   <vm-name>
virsh destroy  <vm-name>   # forced power off (does NOT delete the disk)
