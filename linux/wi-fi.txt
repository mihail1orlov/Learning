# Wi-Fi from the CLI with NetworkManager

# Reference:
#   https://ubuntu.com/core/docs/networkmanager/configure-wifi-connections

## Install

sudo apt install network-manager

## Inspect and connect

nmcli d                        # all network devices
nmcli r wifi on                # turn the radio on
nmcli d wifi list              # scan for SSIDs

# Connect to a network
sudo nmcli d wifi connect <ssid> password <passphrase>
