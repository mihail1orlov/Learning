# vim — handy keystrokes

# Yank (copy) and put
yy            # copy current line
p             # paste below cursor

# Delete
dd            # delete current line
0dG           # delete from current line to end of file

# Indent
>>            # indent current line
<<            # outdent current line

# Visual selection
vip           # select current paragraph (inside)
vap           # select current paragraph (around, includes blank line)
